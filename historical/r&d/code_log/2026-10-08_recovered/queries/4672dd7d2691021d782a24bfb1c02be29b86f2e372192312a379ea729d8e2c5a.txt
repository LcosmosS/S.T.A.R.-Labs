SELECT ALL
    s.specObjID AS SpecObj,
    COALESCE(
        m.OH_Re_fit_O3N2,
        CASE 
            WHEN m.log_OIII_Hb_cen IS NOT NULL AND m.log_NII_Ha_cen IS NOT NULL 
                 AND m.log_OIII_Hb_cen != 0 AND m.log_NII_Ha_cen != 0 
            THEN 8.533 - 0.214 * (m.log_OIII_Hb_cen - m.log_NII_Ha_cen)
            ELSE NULL
        END,
        CASE 
            WHEN e.Flux_OIII_5006 > 0 AND e.Flux_Hb_4861 > 0 
                 AND e.Flux_NII_6583 > 0 AND e.Flux_Ha_6562 > 0
            THEN 8.533 - 0.214 * (
                LOG10(e.Flux_OIII_5006 / e.Flux_Hb_4861) - 
                LOG10(e.Flux_NII_6583 / e.Flux_Ha_6562)
            )
            ELSE NULL
        END
    ) AS OH_O3N2_cen,
    m.OH_Re_fit_T04 AS OH_T04_cen,
    m.Av_gas_Re,
    COALESCE(m.sigma_cen_Ha, e.Sigma_Ha_6562) AS vel_disp_Ha_cen,
    CASE 
        WHEN m.log_Mass IS NOT NULL AND m.re_kpc > 0 
        THEN m.log_Mass - 2 * LOG10(m.re_kpc)
        ELSE NULL
    END AS Sigma_Mass_Re,
    m.Age_LW_Re_fit,
    m.ZH_LW_Re_fit,
    COALESCE(m.EW_Ha_cen, e.EW_Ha_6562) AS EW_Ha_cen,
    m.log_Mass AS stellar_mass,
    m.re_kpc AS Re_kpc,
    COALESCE(m.redshift, s.z) AS redshift
INTO MyDB.DR18_env
FROM SpecObj AS s
LEFT JOIN emissionLinesPort AS e
    ON s.specObjID = e.specObjID
LEFT JOIN mangaPipe3D AS m
    ON ABS(s.ra - m.objra) < 1.0/3600.0 
    AND ABS(s.dec - m.objdec) < 1.0/3600.0
WHERE 
    (COALESCE(m.EW_Ha_cen, e.EW_Ha_6562) > 0)
    AND s.sciencePrimary = 1
    AND s.class = 'GALAXY'

 g.dr7objid = s.objid
 and
 
 s.z < 0.06 and s.z > 0.01

p.objid,p.ra,p.dec,p.u,p.g,p.r,p.i,p.z,
p.run, p.rerun, p.camcol, p.field,
s.specobjid, s.class, s.z as redshift,

  AND g BETWEEN 0 AND 20
