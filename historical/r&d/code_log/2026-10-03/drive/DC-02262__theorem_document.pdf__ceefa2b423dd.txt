primes = [p for p in range(2, prime_bound) if is_prime(p)]
entropy = 0
for p in primes:
if E.has_good_reduction(p):
ap = E.ap(p)
weight = 1/p # Simple weight function
if ap != 0:
entropy -= np.log(abs(ap)) * weight
