  for (k = 1, bins,
    /* Determine bin boundaries */
    start = if(k == 1, 1, end_indices[k-1] + 1);
    end = end_indices[k];
    bin_M = mass_sorted[start..end];     /* Extract bin */
    M0_k = vecmedian(bin_M);             /* Median mass of the bin */
    /* Compute sums for normalization constant K */
    sum_M_over_M0 = sum(i = 1, #bin_M, bin_M[i] / M0_k);
    sum_M0_over_M = sum(i = 1, #bin_M, M0_k / bin_M[i]);
    K = sum_M_over_M0 / sum_M0_over_M;   /* Normalization constant */
    /* Increment rank if K is close to 1 (within 0.2) */
    if (abs(K - 1) < 0.2, rank = rank + 1);
    /* Compute L-function component for this bin */
    L_k = sum(i = 1, #bin_M, (M0_k / bin_M[i])^s);
    L_cosmo = L_cosmo * L_k;             /* Multiply into total L-function */
