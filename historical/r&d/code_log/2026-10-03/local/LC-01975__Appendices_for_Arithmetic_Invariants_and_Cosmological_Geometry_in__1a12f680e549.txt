#  Symbolic  derivation  
r,  R,  Omega,  T,  alpha  =  sp.symbols('r  R  Omega  T  alpha')  
psi  =  r  *  (r  +  1)  /  2  
V  =  (Omega  *  R**(1/r)  *  alpha**r  /  T)  *  sp.exp(-psi)  
print("Universal  ψ( r):",  psi)  
print("Invariant  V:",  V.simplify())
