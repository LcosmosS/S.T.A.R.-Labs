    df['mag_ratio'] = df.get('gmag', 0) / (df.get('rmag', 1) + 1e-8)

    else:
        df['Tully_Fisher'] = df.get('gmag', 0)
        df['T_cosmo'] = 1.0 / (1.0 + df.get('zphot', 0))

        df['betti_ratio'] = df['local_betti_1'] / (df['local_betti_0'] + 1e-8)
