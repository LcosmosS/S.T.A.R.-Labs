  sorted_vec = vecsort(vec);
  Q1_index = floor(0.25 * (n + 1));
  Q3_index = floor(0.75 * (n + 1));
  Q1 = sorted_vec[Q1_index];
  Q3 = sorted_vec[Q3_index];
  IQR = Q3 - Q1;
  lower_bound = Q1 - 1.5 * IQR;
  upper_bound = Q3 + 1.5 * IQR;
  select(x -> x >= lower_bound && x <= upper_bound, vec)
}
